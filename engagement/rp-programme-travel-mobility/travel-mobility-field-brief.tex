\vspace{4pt}
\card{A}{WHO?}{Can the traveller see—and the system prove—which service is asking?}{%
\textbf{Try:} access certificate and registration identify different services.\par
\textbf{Keep:} registered/authenticated service IDs, certificate and registration versions, comparison verdict.\par
\textbf{Limit:} identity binding does not prove necessity or lawfulness.}
\hfill
\card{B}{WHAT / WHY?}{Are the requested facts the minimum needed for this service?}{%
\textbf{Try:} request raw birth date when an age predicate suffices, or an attribute outside registered intended use.\par
\textbf{Keep:} intended-use ID, registered set, requested set, subset/purpose verdict.\par
\textbf{Limit:} a subset check does not decide legal basis.}

\vspace{3pt}
\card{C}{THIS SESSION?}{Is the wallet answer fresh and bound to this transaction?}{%
\textbf{Try:} replay with another audience, nonce, origin, holder or evaluation time.\par
\textbf{Keep:} request/response hashes, challenge and audience commitments, binding method, time, raw reason.\par
\textbf{Limit:} presentation validity does not prove later use stayed within approval.}
\hfill
\card{D}{WHAT HAPPENED?}{Can a later reviewer reconstruct the decision and lifecycle?}{%
\textbf{Try:} mark a deletion request completed when the wallet only opened a browser or email client.\par
\textbf{Keep:} verifier/policy commitment, consumed-attribute commitment, decision reason/time, lifecycle evidence.\par
\textbf{Limit:} a receipt is not a compliance certificate or legal finding.}

\begin{itemize}
  \item \textbf{3 min} — choose one real service journey and its relying decision.
  \item \textbf{4 min} — select the failure that is hardest or most costly to diagnose.
  \item \textbf{5 min} — walk through one synthetic negative case and required evidence.
  \item \textbf{3 min} — agree a technical observation, or end unrecorded.
\end{itemize}
